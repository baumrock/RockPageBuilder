<?php namespace RMBlock;

use RockMatrix\Block;

class {name} extends Block {

  const prefix = "rmblock_{namelower}_";

  public function info() {
    return [
      'icon' => 'picture-o',
      'title' => '{name}',
      'description' => 'RockMatrix Block Setup Demo',
    ];
  }

  public function onCreate() {
    $this->title = '';
  }

  /** ##### frontend ##### */

  /** ##### backend ##### */

  public function migrate() {
    parent::migrate();
    $rm = $this->rm();
    $rm->migrate([
      'fields' => [
      ],
      'templates' => [
        $this->getTplName() => [
          'fields' => [
            'title' => [
              'required' => false,
            ],
          ],
        ],
      ],
    ]);
  }

}
