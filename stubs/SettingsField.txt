public function settingsTable(\ProcessWire\RockFieldsField $field) {
  $settings = $this->getDefaultSettings($field);
  // you can access this setting in your block like this:
  // $block->settings('blockpadding')
  $settings->add([
    'name' => 'blockpadding',
    'label' => 'Block-Padding',
    'value' => $field->input('blockpadding', 'radios', [
      '*s' => 'small padding',
      'm' => 'medium padding',
      'l' => 'large padding',
    ]),
  ]);
  return $settings;
}
